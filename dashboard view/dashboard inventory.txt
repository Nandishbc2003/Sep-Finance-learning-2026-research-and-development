# 2. Main Category Analysis Sheet (Base for Charts 1-4)

    ws_an.append([cat])

for i, cat in enumerate(cats, start=2):
    r = i
    ws_an.cell(r,2, f'=COUNTIF(Data!C2:C{last_row},A{r})')
    ws_an.cell(r,3, f'=SUMIF(Data!C2:C{last_row},A{r},Data!D2:D{last_row})')

    lbl_cell.font = kpi_label_font
    lbl_cell.alignment = Alignment(horizontal="center", vertical="bottom")

    val_cell.number_format = num_format
    val_cell.alignment = Alignment(horizontal="center", vertical="top")
